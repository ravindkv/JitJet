% The soft and the invisible. Left: a pi0 decays into two photons; in the
% laboratory the energy of either photon is uniform between E(1-beta)/2 and
% E(1+beta)/2, so a fraction of the photons of every pi0 falls below the 40 MeV
% of MinPCut. Right: a neutrino passes the cuts of Generator.cc, becomes a
% primary, and the StackingAction kills it before its first step
% (TrackNeutrino = False): the missing transverse momentum of the event is
% decided here.
% ---------------------------------------------------------------------------
\begin{tikzpicture}[x=1cm,y=1cm, every node/.style={font=\scriptsize}, >={Stealth[length=2mm]}]
% ---- (a) pi0 -> gamma gamma
\node[font=\small\bfseries,anchor=west] at (-0.3,2.6) {(a) $\pi^0\to\gamma\gamma$ and the 40 MeV cut};
\fill (0,0.8) circle (2pt);
\draw[->,thick] (-1.2,0.8) -- (-0.05,0.8) node[midway,above] {$\pi^0$, $E$};
\draw[->,thick,green!50!black,decorate,decoration={snake,amplitude=1.2pt,segment length=6pt,post length=2mm}] (0,0.8) -- (2.4,1.25) node[right] {$E_1=\tfrac{E}{2}(1+\beta\cos\theta^*)$};
\draw[->,thick,green!50!black,decorate,decoration={snake,amplitude=1.2pt,segment length=6pt,post length=2mm}] (0,0.8) -- (-0.45,0.05) node[below] {$E_2=\tfrac{E}{2}(1-\beta\cos\theta^*)$};
% the flat spectrum
\begin{scope}[shift={(-0.3,-2.5)}]
\draw[->] (0,0) -- (3.6,0) node[right] {$E_\gamma$};
\draw[->] (0,0) -- (0,1.3) node[left] {$\mathrm{d}N/\mathrm{d}E_\gamma$};

\node[anchor=west,text width=3.7cm,align=left] at (3.9,0.55) {uniform: $P(E_\gamma<E_c)=\dfrac{E_c-(E-p)/2}{p}\simeq\dfrac{E_c}{E}$};

% ---- (b) the neutrino
